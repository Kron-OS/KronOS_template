# Source: kronos/templates/networkpolicies/deny-app-egress.yaml
# Default-deny egress from App zone; only allow to Data zone and DNS.
apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: kronos-deny-app-internet-egress
  namespace: kronos
  labels:
    helm.sh/chart: kronos-0.1.0
    app.kubernetes.io/name: kronos
    app.kubernetes.io/instance: kronos
    app.kubernetes.io/version: "0.1.0"
    app.kubernetes.io/managed-by: Helm
spec:
  podSelector:
    matchLabels:
      app.kubernetes.io/zone: app
  policyTypes:
    - Egress
  egress:
    # Allow DNS resolution — scoped to kube-dns only (INFRA-009). The
    # previous rule had no `to:` selector at all, letting App-zone pods
    # resolve DNS against any destination on port 53/UDP+TCP, a potential
    # exfiltration/tunneling channel. kube-system + k8s-app: kube-dns is the
    # standard label convention for CoreDNS/kube-dns on EKS/GKE/AKS/kubeadm;
    # adjust the namespaceSelector/podSelector if your CNI/cluster uses a
    # different DNS provider or labeling convention.
    - to:
        - namespaceSelector:
            matchLabels:
              kubernetes.io/metadata.name: kube-system
          podSelector:
            matchLabels:
              k8s-app: kube-dns
      ports:
        - protocol: UDP
          port: 53
        - protocol: TCP
          port: 53
    # Allow traffic to Data zone only
    - to:
        - podSelector:
            matchLabels:
              app.kubernetes.io/zone: data
    # Allow traffic to Keycloak (auth zone — same namespace)
    - to:
        - podSelector:
            matchLabels:
              app.kubernetes.io/component: keycloak

# Source: kronos/templates/networkpolicies/parser-egress.yaml
# INFRA-008: fast/plaso parser pods previously shared the `zone: app` label
# with the backend/celery-index/celery-beat workloads, which meant they
# inherited deny-app-egress.yaml's "App zone -> entire Data zone" allowance
# (Postgres, Redis, OpenSearch, Vault, KES, Keycloak) even though parsers
# only ever need to read/write evidence objects in MinIO. Per
# Project_Specifications.md §5.4, parser sandboxes should have minimal
# network reachability as defense-in-depth on top of the sandbox itself
# running with network=none internally.
#
# celery/deployment-fast.yaml and celery/deployment-plaso.yaml now label
# their pods `app.kubernetes.io/zone: app-parser` instead, so they no longer
# match deny-app-egress.yaml's `zone: app` selector, and this policy becomes
# their only source of egress: DNS (see the kube-dns note in
# deny-app-egress.yaml for the INFRA-009 convention) plus MinIO.
#
# NOTE: MinIO is not deployed by this chart (see values.yaml `minio.endpoint`,
# an external/pre-existing service) so there is no chart-controlled label to
# select it precisely. `app.kubernetes.io/name: minio` matches the selector
# labels the official/Bitnami MinIO charts use — verify this against
# whatever manifest actually deploys MinIO in your cluster and adjust if it
# differs.
apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: kronos-parser-egress
  namespace: kronos
  labels:
    helm.sh/chart: kronos-0.1.0
    app.kubernetes.io/name: kronos
    app.kubernetes.io/instance: kronos
    app.kubernetes.io/version: "0.1.0"
    app.kubernetes.io/managed-by: Helm
spec:
  podSelector:
    matchLabels:
      app.kubernetes.io/zone: app-parser
  policyTypes:
    - Egress
  egress:
    # DNS resolution — scoped to kube-dns only, not "anywhere" (INFRA-009).
    - to:
        - namespaceSelector:
            matchLabels:
              kubernetes.io/metadata.name: kube-system
          podSelector:
            matchLabels:
              k8s-app: kube-dns
      ports:
        - protocol: UDP
          port: 53
        - protocol: TCP
          port: 53
    # MinIO only — parser pods stream evidence in/out and do nothing else.
    - to:
        - podSelector:
            matchLabels:
              app.kubernetes.io/name: minio
      ports:
        - protocol: TCP
          port: 9000

    #!/bin/bash

    timeout=$1
    run_redis_command() {
        local args=("-p" "$REDIS_PORT")
        timeout -s 15 $timeout redis-cli "${args[@]}" "$@"
    }

    [[ -f $REDIS_PASSWORD_FILE ]] && export REDIS_PASSWORD="$(< "${REDIS_PASSWORD_FILE}")"
    [[ -n "$REDIS_PASSWORD" ]] && export REDISCLI_AUTH="$REDIS_PASSWORD"
    response=$(run_redis_command ping)
    if [ "$?" -eq "124" ]; then
      echo "Timed out"
      exit 1
    fi
    if [ "$response" != "PONG" ]; then
      echo "$response"
      exit 1
    fi

    #!/bin/bash

    [[ -f $REDIS_PASSWORD_FILE ]] && export REDIS_PASSWORD="$(< "${REDIS_PASSWORD_FILE}")"
    [[ -n "$REDIS_PASSWORD" ]] && export REDISCLI_AUTH="$REDIS_PASSWORD"
    response=$(
      timeout -s 15 $1 \

        -h localhost \

      echo "$response"

    #!/bin/bash

    [[ -f $REDIS_PASSWORD_FILE ]] && export REDIS_PASSWORD="$(< "${REDIS_PASSWORD_FILE}")"
    if [[ -f /opt/bitnami/redis/mounted-etc/master.conf ]];then
        cp /opt/bitnami/redis/mounted-etc/master.conf /opt/bitnami/redis/etc/master.conf
    fi
    if [[ -f /opt/bitnami/redis/mounted-etc/redis.conf ]];then
        cp /opt/bitnami/redis/mounted-etc/redis.conf /opt/bitnami/redis/etc/redis.conf
    fi
    if [[ -f /opt/bitnami/redis/mounted-etc/users.acl ]];then
        cp /opt/bitnami/redis/mounted-etc/users.acl /opt/bitnami/redis/etc/users.acl
    fi
    ARGS=("--port" "${REDIS_PORT}")
    ARGS+=("--requirepass" "${REDIS_PASSWORD}")
    ARGS+=("--masterauth" "${REDIS_PASSWORD}")
    ARGS+=("--include" "/opt/bitnami/redis/etc/redis.conf")
    ARGS+=("--include" "/opt/bitnami/redis/etc/master.conf")
    exec redis-server "${ARGS[@]}"

        hostname="$1"
        type="$2"

        if [ -z "$port" ]; then
            case $type in
                "SENTINEL")
                    echo 26379
                    ;;
                "REDIS")
                    echo 6379
                    ;;
            esac
        else
            echo $port
        fi
    }

    get_full_hostname() {
        hostname="$1"
        full_hostname="${hostname}.${HEADLESS_SERVICE}"
        echo "${full_hostname}"
    }

    REDISPORT=$(get_port "$HOSTNAME" "REDIS")
    HEADLESS_SERVICE="kronos-redis-auth-streams-headless.default.svc.cluster.local"

    [[ -f $REDIS_PASSWORD_FILE ]] && export REDIS_PASSWORD="$(< "${REDIS_PASSWORD_FILE}")"
    [[ -f $REDIS_MASTER_PASSWORD_FILE ]] && export REDIS_MASTER_PASSWORD="$(< "${REDIS_MASTER_PASSWORD_FILE}")"
    if [[ -f /opt/bitnami/redis/mounted-etc/replica.conf ]];then
        cp /opt/bitnami/redis/mounted-etc/replica.conf /opt/bitnami/redis/etc/replica.conf

    fi

# Source: kronos/templates/backend/migrate-job.yaml
# Pre-install/pre-upgrade hook that runs `alembic upgrade head` against the
# real Postgres schema before the backend Deployment (and Celery workers) is
# ever created (install) or updated (upgrade). This is the Helm/Kubernetes
# equivalent of docker-compose.{dev,test,prod}.yml's own `db-migrate`
# one-shot service (see docker-compose.dev.yml and docs/DATABASE_MIGRATIONS.md)
# -- Milestone V4 (Gap Audit P1-12) replaced the old create_tables()-at-boot
# path used by wire_dependencies_async()/wire_dependencies_sync() with a
# real Alembic-managed schema, but that change only ever touched the Compose
# files, not this chart. Helm has no `depends_on: service_completed_successfully`
# construct, so a hook Job blocking the whole install/upgrade is the closest
# equivalent. Without this, a fresh install reports every pod Ready
# (src/external/fastapi_app.py's /healthz is deliberately DB-independent --
# do NOT add a DB check there, see its own comment: a transient downstream
# outage must never cause Kubernetes to kill an otherwise-healthy backend)
# against an empty schema, with zero error signal from Kubernetes -- every
# DB-touching request just 500s.
#
# Uses the same backend image: alembic.ini/migrations/ are baked into it
# (see docker/Dockerfile's own "Alembic (Gap Audit P1-12 / Milestone V4)"
# comment), no separate image/Dockerfile needed, same as db-migrate's own
# `image: kronos-backend:dev` in docker-compose.dev.yml. Loads DATABASE_URL
# from the exact same kronos-app-secrets secret the backend Deployment
# itself uses (charts/kronos/templates/backend/deployment.yaml) -- no new
# secret name or env var invented.
#
# hook-weight "0": lower than keycloak.provisionOrg / opensearch.securityInit's
# post-install hook-weight "5" above -- though those run in a different hook
# phase entirely (post-install/post-upgrade, after every resource including
# the backend Deployment already exists), so this weight mostly matters
# relative to any other pre-install/pre-upgrade hook added later.
apiVersion: batch/v1
kind: Job
metadata:
  name: kronos-db-migrate
  namespace: kronos
  labels:
    helm.sh/chart: kronos-0.1.0
    app.kubernetes.io/name: kronos
    app.kubernetes.io/instance: kronos
    app.kubernetes.io/version: "0.1.0"
    app.kubernetes.io/managed-by: Helm
    app.kubernetes.io/component: db-migrate
  annotations:
    "helm.sh/hook": pre-install,pre-upgrade
    "helm.sh/hook-weight": "0"
    "helm.sh/hook-delete-policy": before-hook-creation,hook-succeeded
spec:
  backoffLimit: 3
  template:
    metadata:
      labels:
        app.kubernetes.io/name: kronos
        app.kubernetes.io/instance: kronos
        app.kubernetes.io/component: db-migrate
        app.kubernetes.io/zone: app
    spec:
      serviceAccountName: kronos
      automountServiceAccountToken: false
      restartPolicy: Never
      securityContext:
        runAsNonRoot: true
        runAsUser: 1000
        fsGroup: 1000
        seccompProfile:
          type: RuntimeDefault
      containers:
        - name: db-migrate
          image: "ghcr.io/kron-os/kronos-backend:0.1.0"
          imagePullPolicy: IfNotPresent
          command: ["alembic", "upgrade", "head"]
          envFrom:
            - secretRef:
                name: kronos-app-secrets
          securityContext:
            allowPrivilegeEscalation: false
            readOnlyRootFilesystem: true
            capabilities:
              drop: ["ALL"]
          volumeMounts:
            - name: tmp
              mountPath: /tmp
      volumes:
        - name: tmp
          emptyDir: {}
